# ES 索引模板与 Kibana 语法

## 一、索引模板（Index Template）

```json
PUT _index_template/infra_oss_files_template
{
  "index_patterns": ["infra_oss_files*"],
  "template": {
    "settings": {
      "number_of_shards": 3,
      "number_of_replicas": 3,
      "index.max_result_window": 10000
    },
    "mappings": {
      "properties": {
        "fileId": {
          "type": "keyword"
        },
        "objectKey": {
          "type": "keyword"
        },
        "bucket": {
          "type": "keyword"
        },
        "visibility": {
          "type": "keyword"
        },
        "fileType": {
          "type": "keyword"
        },
        "fileName": {
          "type": "text",
          "fields": {
            "keyword": {
              "type": "keyword",
              "ignore_above": 256
            }
          }
        },
        "contentType": {
          "type": "keyword"
        },
        "fileSize": {
          "type": "long"
        },
        "userId": {
          "type": "keyword"
        },
        "uploadTime": {
          "type": "long"
        },
        "expireTime": {
          "type": "long"
        },
        "status": {
          "type": "keyword"
        }
      }
    }
  },
  "priority": 500,
  "version": 1,
  "_meta": {
    "description": "OSS 文件元数据索引模板"
  }
}
```

### 模板说明

- `index_patterns`: `infra_oss_files*` 匹配所有以 `infra_oss_files` 开头的索引
- `number_of_shards`: 3 个主分片，适合中等数据量
- `dynamic: strict`: 拒绝未定义字段，防止脏数据写入
- `fileName`: 使用 `ik_max_word` 中文分词，同时保留 `keyword` 子字段
- `uploadTime` / `expireTime`: 数值类型（毫秒时间戳），便于范围查询和排序

---

## 二、创建索引

```json
PUT /infra_oss_files
```

创建后会自动应用上述模板。如需按时间滚动（如 `infra_oss_files-2026.05`），直接创建对应名称索引即可。

## 三、文档写入语法

### 3.1 生成预签名时写入（PENDING 状态）

```json
PUT /infra_oss_files/_doc/file-uuid-001
{
  "fileId": "file-uuid-001",
  "objectKey": "images/user12345/2026-05-09/a1b2c3d4-avatar.jpg",
  "bucket": "liko-public-bucket",
  "visibility": "PUBLIC",
  "fileType": "IMAGE",
  "fileName": "avatar.jpg",
  "contentType": "image/jpeg",
  "fileSize": 0,
  "userId": "12345",
  "uploadTime": null,
  "expireTime": 1746793800000,
  "status": "PENDING"
}
```

### 3.2 上传完成后更新（COMPLETED 状态）

```json
POST /infra_oss_files/_update/file-uuid-001
{
  "doc": {
    "fileSize": 1048576,
    "uploadTime": 1746792900000,
    "status": "COMPLETED"
  }
}
```

### 3.3 批量写入（Bulk）

```json
POST /infra_oss_files/_bulk
{ "index": { "_id": "file-uuid-002" } }
{ "fileId": "file-uuid-002", "objectKey": "documents/user67890/2026-05-09/uuid-report.pdf", "bucket": "liko-private-bucket", "visibility": "PRIVATE", "fileType": "DOCUMENT", "fileName": "report.pdf", "contentType": "application/pdf", "fileSize": 0, "userId": "67890", "uploadTime": null, "expireTime": 1746795600000, "status": "PENDING" }
{ "index": { "_id": "file-uuid-003" } }
{ "fileId": "file-uuid-003", "objectKey": "videos/user11111/2026-05-09/uuid-demo.mp4", "bucket": "liko-private-bucket", "visibility": "PRIVATE", "fileType": "VIDEO", "fileName": "demo.mp4", "contentType": "video/mp4", "fileSize": 52428800, "userId": "11111", "uploadTime": 1746788400000, "expireTime": 1746790200000, "status": "COMPLETED" }
```

## 四、常用查询语法

### 4.1 查询用户文件列表（分页）

```json
GET /infra_oss_files/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "userId": "12345" } },
        { "term": { "status": "COMPLETED" } }
      ]
    }
  },
  "sort": [
    { "uploadTime": { "order": "desc" } }
  ],
  "from": 0,
  "size": 20
}
```

### 4.2 按文件名搜索（全文检索）

```json
GET /infra_oss_files/_search
{
  "query": {
    "bool": {
      "must": [
        { "match": { "fileName": "年度报告" } },
        { "term": { "userId": "12345" } }
      ]
    }
  }
}
```

### 4.3 按类型 + 时间范围筛选

```json
GET /infra_oss_files/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "fileType": "IMAGE" } },
        { "term": { "visibility": "PUBLIC" } },
        { "term": { "status": "COMPLETED" } },
        {
          "range": {
            "uploadTime": {
              "gte": 1746057600000,
              "lte": 1746815999999
            }
          }
        }
      ]
    }
  }
}
```

### 4.4 用户存储用量统计

```json
GET /infra_oss_files/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "userId": "12345" } },
        { "term": { "status": "COMPLETED" } }
      ]
    }
  },
  "aggs": {
    "totalSize": {
      "sum": { "field": "fileSize" }
    },
    "typeDistribution": {
      "terms": { "field": "fileType" }
    }
  },
  "size": 0
}
```

### 4.5 查询 PENDING 且已超期的记录（定时清理用）

```json
GET /infra_oss_files/_search
{
  "query": {
    "bool": {
      "must": [
        { "term": { "status": "PENDING" } },
        {
          "range": {
            "expireTime": {
              "lt": 1746796800000
            }
          }
        }
      ]
    }
  }
}
```

### 4.6 更新超期 PENDING 为 EXPIRED（示例时间戳需替换为当前时间）

```json
POST /infra_oss_files/_update_by_query
{
  "script": {
    "source": "ctx._source.status = 'EXPIRED'",
    "lang": "painless"
  },
  "query": {
    "bool": {
      "must": [
        { "term": { "status": "PENDING" } },
        {
          "range": {
            "expireTime": {
              "lt": 1746796800000
            }
          }
        }
      ]
    }
  }
}
```

## 五、删除语法

### 5.1 删除单条记录

```json
DELETE /infra_oss_files/_doc/file-uuid-001
```

### 5.2 按条件删除（如用户注销后清理）

```json
POST /infra_oss_files/_delete_by_query
{
  "query": {
    "term": { "userId": "12345" }
  }
}
```
